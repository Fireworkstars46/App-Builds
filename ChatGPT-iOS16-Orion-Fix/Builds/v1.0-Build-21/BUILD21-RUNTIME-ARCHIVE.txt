(() => {
  "use strict";

  // Build 21: narrowly scoped ChatGPT browser-version compatibility gate override.
  // Keep the real iPhone/WebKit identity while presenting a currently supported
  // iOS/Safari version to client-side browser-version gates. Build 21 also configures a Safari/WebExtension declarative network rule to present the same identity in the HTTP User-Agent request header for chatgpt.com only. No other request headers are changed.
  if (location.origin !== "https://chatgpt.com") return;

  const STATUS_KEY = "__chatgptIOS16UaCompatStatus";
  const TARGET_UA = "Mozilla/5.0 (iPhone; CPU iPhone OS 18_7 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/18.7 Mobile/15E148 Safari/604.1";
  const TARGET_APP_VERSION = "5.0 (iPhone; CPU iPhone OS 18_7 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/18.7 Mobile/15E148 Safari/604.1";

  const original = {
    userAgent: safeString(() => navigator.userAgent),
    appVersion: safeString(() => navigator.appVersion)
  };

  function safeString(read) {
    try { return String(read() || ""); }
    catch (_) { return ""; }
  }

  function installGetter(name, value) {
    try {
      Object.defineProperty(navigator, name, {
        configurable: true,
        enumerable: true,
        get: () => value
      });
      if (safeString(() => navigator[name]) === value) return "instance";
    } catch (_) {}

    try {
      const proto = Object.getPrototypeOf(navigator);
      if (!proto) return "failed";
      const current = Object.getOwnPropertyDescriptor(proto, name);
      if (current && current.configurable === false) return "failed";
      Object.defineProperty(proto, name, {
        configurable: true,
        enumerable: current ? Boolean(current.enumerable) : true,
        get: () => value
      });
      if (safeString(() => navigator[name]) === value) return "prototype";
    } catch (_) {}

    return "failed";
  }

  const userAgentMode = installGetter("userAgent", TARGET_UA);
  const appVersionMode = installGetter("appVersion", TARGET_APP_VERSION);

  const status = Object.freeze({
    version: "1.0",
    build: 21,
    active: userAgentMode !== "failed" && appVersionMode !== "failed",
    originalUserAgent: original.userAgent,
    originalAppVersion: original.appVersion,
    effectiveUserAgent: safeString(() => navigator.userAgent),
    effectiveAppVersion: safeString(() => navigator.appVersion),
    userAgentMode,
    appVersionMode,
    target: "iOS/Safari 18.7",
    httpUserAgentOverrideConfigured: true,
    httpUserAgentTarget: TARGET_UA
  });

  try {
    Object.defineProperty(window, STATUS_KEY, {
      configurable: true,
      value: status
    });
  } catch (_) {
    try { window[STATUS_KEY] = status; } catch (_) {}
  }
})();
