    private static T? FindControl<T>(Control root, Func<T, bool> predicate) where T : Control
    {
        foreach (Control child in root.Controls)
        {
            if (child is T typed && predicate(typed)) return typed;
            var nested = FindControl<T>(child, predicate);
            if (nested != null) return nested;
        }
        return null;
    }

    private static bool ContainsControlText(Control root, string text)
    {
        foreach (Control child in root.Controls)
        {
            if (string.Equals(child.Text, text, StringComparison.OrdinalIgnoreCase)) return true;
            if (ContainsControlText(child, text)) return true;
        }
        return false;
